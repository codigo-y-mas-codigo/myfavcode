 <script>
  const textInputs = [
   document.getElementById('textInput1'),
   document.getElementById('textInput2'),
   document.getElementById('textInput3')
  ];
  const wordCounts = [
   document.getElementById('wordCount1'),
   document.getElementById('wordCount2'),
   document.getElementById('wordCount3')
  ];
  const totalWords = document.getElementById('totalWords');
  const averageWords = document.getElementById('averageWords');
  const wordInfo = document.getElementById('wordInfo');
  const toast = document.getElementById('toast');
  const currentYear = document.getElementById('currentYear');
  const contactModal = document.getElementById('contactModal');
  const contactBtn = document.getElementById('contactBtn');
  const closeModal = document.getElementById('closeModal');
  currentYear.textContent = new Date().getFullYear();
  textInputs.forEach((input, index) => {
   input.addEventListener('input', () => {
    const text = input.value.trim();
    const words = text ? text.split(/\s+/) : [];
    wordCounts[index].textContent = `Palabras: ${words.length}`;
    updateTotals();
   });
  });

  function updateTotals() {
   let total = 0;
   let filledContainers = 0;
   let minWords = Infinity;
   let maxWords = 0;
   textInputs.forEach((input) => {
    const text = input.value.trim();
    const words = text ? text.split(/\s+/) : [];
    const wordCount = words.length;
    total += wordCount;
    if (wordCount > 0) {
     filledContainers++;
     minWords = Math.min(minWords, wordCount);
     maxWords = Math.max(maxWords, wordCount);
    }
   });
   totalWords.textContent = `Total Palabras: ${total}`;
   const average = filledContainers > 0 ? (total / filledContainers).toFixed(2) : 0;
   averageWords.textContent = `Promedio Palabras: ${average}`;
   // Actualizar el texto informativo
   wordInfo.textContent = `Desarrolla el contenido entre un mínimo de ${minWords + 10} y un máximo de ${maxWords + 10} palabras.`;
   wordInfo.onmouseup = () => copyToClipboard(wordInfo.textContent);
  }

  function copyToClipboard(text) {
   navigator.clipboard.writeText(text).then(() => {
    showToast();
   }).catch(err => {
    console.error('Error al copiar: ', err);
   });
  }

  function showToast() {
   toast.className = "show";
   setTimeout(() => {
    toast.className = toast.className.replace("show", "");
   }, 3000);
  }
  contactBtn.onclick = function() {
   contactModal.style.display = "flex";
  }
  closeModal.onclick = function() {
   contactModal.style.display = "none";
  }
  window.onclick = function(event) {
   if (event.target === contactModal) {
    contactModal.style.display = "none";
   }
  }
 </script>
