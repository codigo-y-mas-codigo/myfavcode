  <script src='//cdnjs.cloudflare.com/ajax/libs/jquery/2.2.2/jquery.min.js'></script>
  <script>
    // Seleccionamos el elemento cuerpo una vez que el DOM está completamente cargado
    document.addEventListener("DOMContentLoaded", function() {
      const body = document.body;

      // Usamos addEventListener para vincular el evento mousemove
      body.addEventListener("mousemove", function(event) {
        // Obtenemos las coordenadas del ratón
        const mouseX = event.pageX;
        const mouseY = event.pageY;

        // Obtenemos el elemento skeleton
        const skeleton = document.querySelector(".skeleton");

        // Calculamos las nuevas posiciones del skeleton para que siga al ratón
        const newLeft = mouseX - skeleton.offsetWidth / 2;
        const newTop = mouseY - skeleton.offsetHeight / 2;

        // Aplicamos las nuevas posiciones al skeleton
        skeleton.style.left = newLeft + "px";
        skeleton.style.top = newTop + "px";
      });
    });
  </script>
