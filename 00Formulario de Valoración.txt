 <script>
  document.addEventListener("DOMContentLoaded", () => {
   const form = document.getElementById("gform");
   const submitBtn = document.getElementById("feedback-submit-button");
   const btnText = document.getElementById("button-text");
   const waitMsg = document.getElementById("feedback-wait-message");
   const thankYouModal = document.getElementById("thankyouModal");
   const errorMsg = document.getElementById("captcha-error");
   let captchaValidado = false;
   // Lógica de los iconos interactivos
   const iconos = document.querySelectorAll('.icono-opcion');
   iconos.forEach(icono => {
    icono.addEventListener('click', (e) => {
     // Limpiar selecciones previas
     iconos.forEach(i => i.classList.remove('activo'));
     // Activar el seleccionado
     e.target.classList.add('activo');
     // Comprobar si es el correcto
     if (e.target.getAttribute('data-val') === 'verdadero') {
      captchaValidado = true;
      errorMsg.classList.add("feedback-hidden");
     } else {
      captchaValidado = false;
     }
    });
   });
   form.addEventListener("submit", async (e) => {
    e.preventDefault();
    // Validación del Anti-Spam antes de hacer Fetch
    if (!captchaValidado) {
     errorMsg.classList.remove("feedback-hidden");
     return;
    }
    const formData = new FormData(form);
    toggleSubmitState(true);
    try {
     await fetch(form.action, {
      method: 'POST',
      body: formData
     });
     thankYouModal.style.display = "block";
     setTimeout(() => thankYouModal.style.display = "none", 3000);
     // Resetear formulario y captcha
     form.reset();
     iconos.forEach(i => i.classList.remove('activo'));
     captchaValidado = false;
    } catch (error) {
     alert("Error de conexión. Intenta de nuevo.");
    } finally {
     toggleSubmitState(false);
    }
   });

   function toggleSubmitState(isSubmitting) {
    submitBtn.disabled = isSubmitting;
    btnText.classList.toggle("feedback-hidden", isSubmitting);
    waitMsg.classList.toggle("feedback-hidden", !isSubmitting);
   }
  });
 </script>
