    <script id="rendered-js" >
var container = document.getElementById("container");
var triangleTotal = 100;
var triangleArray = [];
var triIndex = 0

function getRandom(min, max) {
    return min + Math.random() * (max - min);
}

for (var i = 0; i < triangleTotal; i++) {
    var triangle = document.createElementNS("http://www.w3.org/2000/svg", 'polygon');
    triangle.setAttribute('points', "646 315 647 300 658 310 646 315");

    if (i%2 == 0) triangle.setAttribute('fill', "#1270db");
    else triangle.setAttribute('fill', "#ffffff");

    triangleArray.push(triangle);
    TweenLite.set(triangle, { autoAlpha: 0 });
    container.appendChild(triangle);
}

var tl = new TimelineMax({ delay:0.2, repeat:-1 }).add(triangleRun, 0.02)

function triangleRun() {
    var length = Math.random() * 200;
    var tri = triangleArray[triIndex++]; 
    if (triIndex === triangleTotal) triIndex = 0; 

    TweenLite.set(tri, {autoAlpha:1});
    TweenLite.to(tri, 1.5, {
        rotation:getRandom(0, 360),
        scale:getRandom(0.5, 1.5),
        x:Math.cos(Math.random() * Math.PI * 2) * length * 6, 
        y:Math.sin(Math.random() * Math.PI * 2) * length * 5,

        onComplete: function() {
            TweenLite.set(tri, {x:0, y:0, autoAlpha:0})
        },
    }, 0);
}
