  /* ---- Auto-advance engine synced to a fixed total duration ---- */
  const TOTAL_SECONDS = 374; // 6:14
  const chatBody = document.getElementById('chat-body');
  const trackFill = document.getElementById('trackFill');
  const curTimeEl = document.getElementById('curTime');
  const playBtn = document.getElementById('playBtn');
  const playIcon = document.getElementById('playIcon');
  const track = document.getElementById('track');
  const sidebarPreview = document.getElementById('sidebar-preview');
  const sidebarTime = document.getElementById('sidebar-time');
  // Weight each item by content length so longer lines get more "reading time"
  function weightOf(item) {
   if (item.type === 'card') return 60; // fixed weight, like a pause on an image
   const len = item.text.length;
   return Math.max(18, len);
  }
  const weights = items.map(weightOf);
  const totalWeight = weights.reduce((a, b) => a + b, 0);
  const durations = weights.map(w => (w / totalWeight) * TOTAL_SECONDS * 1000); // ms per item
  let idx = 0;
  let elapsed = 0;
  let playing = false;
  let timer = null;

  function fmt(s) {
   const m = Math.floor(s / 60);
   const sec = Math.floor(s % 60);
   return String(m).padStart(2, '0') + ':' + String(sec).padStart(2, '0');
  }

  function renderItem(item) {
   if (item.type === 'card') {
    const wrap = document.createElement('div');
    wrap.className = 'card-box';
    wrap.innerHTML = `
      <div class="stat-card">
        <div class="stat-title">${item.title}</div>
        <div class="stat-big">${item.big}</div>
        <div class="stat-desc">${item.desc}</div>
      </div>`;
    chatBody.appendChild(wrap);
    sidebarPreview.textContent = '📎 ' + item.title;
   } else {
    const cls = item.speaker === 'V' ? 'msg-v' : 'msg-b';
    const showName = (idx === 0) || (items[idx - 1] && items[idx - 1].speaker !== item.speaker);
    const ticks = item.speaker === 'V' ? '<span class="ticks read"><i class="fa-solid fa-check-double"></i></span>' : '';
    const div = document.createElement('div');
    div.className = 'message-box ' + cls;
    div.innerHTML = `<p>${showName ? `<span class="sender">${item.name}</span>` : ''}${item.text}<span class="meta">${fmt(elapsed/1000)} ${ticks}</span></p>`;
    chatBody.appendChild(div);
    sidebarPreview.textContent = item.name + ': ' + item.text.replace(/<[^>]*>/g, '').slice(0, 40);
   }
   chatBody.scrollTop = chatBody.scrollHeight;
   sidebarTime.textContent = fmt(elapsed / 1000);
  }

  function tick() {
   if (idx >= items.length) {
    playing = false;
    playIcon.className = 'fa-solid fa-rotate-right';
    clearTimeout(timer);
    return;
   }
   renderItem(items[idx]);
   elapsed += durations[idx];
   trackFill.style.width = Math.min(100, (elapsed / (TOTAL_SECONDS * 1000)) * 100) + '%';
   curTimeEl.textContent = fmt(elapsed / 1000);
   idx++;
   if (playing) {
    timer = setTimeout(tick, durations[idx - 1] * 0.55); // pacing: readable but continuous
   }
  }

  function play() {
   if (idx >= items.length) {
    restart();
   }
   playing = true;
   playIcon.className = 'fa-solid fa-pause';
   tick();
  }

  function pause() {
   playing = false;
   playIcon.className = 'fa-solid fa-play';
   clearTimeout(timer);
  }

  function restart() {
   clearTimeout(timer);
   idx = 0;
   elapsed = 0;
   chatBody.innerHTML = '<div class="date-chip">Grabación de hoy</div>';
   trackFill.style.width = '0%';
   curTimeEl.textContent = '00:00';
  }
  playBtn.addEventListener('click', () => {
   if (idx >= items.length) {
    restart();
    play();
    return;
   }
   playing ? pause() : play();
  });
  track.addEventListener('click', (e) => {
   const rect = track.getBoundingClientRect();
   const pct = Math.min(1, Math.max(0, (e.clientX - rect.left) / rect.width));
   const targetElapsed = pct * TOTAL_SECONDS * 1000;
   pause();
   restart();
   // Fast-forward: render every item up to the target time instantly
   while (elapsed < targetElapsed && idx < items.length) {
    renderItem(items[idx]);
    elapsed += durations[idx];
    idx++;
   }
   trackFill.style.width = Math.min(100, (elapsed / (TOTAL_SECONDS * 1000)) * 100) + '%';
   curTimeEl.textContent = fmt(elapsed / 1000);
  });
  // Auto-start the playback so the user just watches it unfold, no scrolling needed
  window.addEventListener('load', () => {
   play();
  });
